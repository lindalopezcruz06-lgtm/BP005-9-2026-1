#include <stdio.h>

   int main() {

       int i; // Variable de control
       int contador = 0; // Cuenta cuantos pares hay

       for (i = 1; i <= 20; i++) {
          if (i % 2 == 0) {
             printf("%d es par\n", i);
             contador++;
          }
       }

       printf("Cantidad de numeros pares: %d\n", contador);

       return 0;

}
