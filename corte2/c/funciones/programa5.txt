#include <stdio.h>

void leerArreglo(int v[], int n);
int sumarArreglo(int v[], int n);
float calcularPromedio(int suma, int n);
void mostrarArreglo(int v[], int n);

int main() {
    int arreglo[5];
    int suma;
    float promedio;

    leerArreglo(arreglo, 5);

    printf("\nElementos del arreglo:\n");
    mostrarArreglo(arreglo, 5);

    suma = sumarArreglo(arreglo, 5);

    promedio = calcularPromedio(suma, 5);

    printf("\nSuma: %d\n", suma);
    printf("Promedio: %.2f\n", promedio);

    return 0;
}

void leerArreglo(int v[], int n) {
    int i;
    for (i = 0; i < n; i++) {
        printf("Digite el elemento %d: ", i);
        scanf("%d", &v[i]);
    }
}

void mostrarArreglo(int v[], int n) {
    int i;
    for (i = 0; i < n; i++) {
        printf("%d ", v[i]);
    }
    printf("\n");
}

int sumarArreglo(int v[], int n) {
    int i, suma = 0;
    for (i = 0; i < n; i++) {
        suma += v[i];
    }
    return suma;
}

float calcularPromedio(int suma, int n) {
    return (float)suma / n;
}
