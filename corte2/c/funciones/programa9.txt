#include <stdio.h>


int factorialIterativo(int n);
int factorialRecursivo(int n);

int main() {
    int numero;
    int factIter, factRec;

    printf("Digite un numero entero no negativo: ");
    scanf("%d", &numero);

    if (numero < 0) {
        printf("Error: el numero debe ser no negativo.\n");
        return 1;
    }

    factIter = factorialIterativo(numero);
    factRec = factorialRecursivo(numero);

    printf("Factorial iterativo: %d\n", factIter);
    printf("Factorial recursivo: %d\n", factRec);

    return 0;
}

int factorialIterativo(int n) {
    int resultado = 1;

    for (int i = 1; i <= n; i++) {
        resultado *= i;
    }

    return resultado;
}

int factorialRecursivo(int n) {
    if (n == 0 || n == 1) {
        return 1;
    } else {
        return n * factorialRecursivo(n - 1);
    }
}
