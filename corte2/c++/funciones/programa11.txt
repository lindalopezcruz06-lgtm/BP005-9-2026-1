#include <iostream>
using namespace std;

#define N 5
void leerNotas(float notas[], int n);
float obtenerMayor(float notas[], int n);
float obtenerMenor(float notas[], int n);
float calcularPromedio(float notas[], int n);
int contarAprobadas(float notas[], int n);
int contarReprobadas(float notas[], int n);
void mostrarReporte(float mayor, float menor, float promedio, int aprobadas, int reprobadas);

int main() {
    float notas[N];

    leerNotas(notas, N);

    float mayor = obtenerMayor(notas, N);
    float menor = obtenerMenor(notas, N);
    float promedio = calcularPromedio(notas, N);
    int aprobadas = contarAprobadas(notas, N);
    int reprobadas = contarReprobadas(notas, N);

    mostrarReporte(mayor, menor, promedio, aprobadas, reprobadas);

    return 0;
}

void leerNotas(float notas[], int n) {
    for (int i = 0; i < n; i++) {
        cout << "Digite la nota " << i + 1 << ": ";
        cin >> notas[i];
    }
}

float obtenerMayor(float notas[], int n) {
    float mayor = notas[0];
    for (int i = 1; i < n; i++) {
        if (notas[i] > mayor) {
            mayor = notas[i];
        }
    }
    return mayor;
}

float obtenerMenor(float notas[], int n) {
    float menor = notas[0];
    for (int i = 1; i < n; i++) {
        if (notas[i] < menor) {
            menor = notas[i];
        }
    }
    return menor;
}

float calcularPromedio(float notas[], int n) {
    float suma = 0;
    for (int i = 0; i < n; i++) {
        suma += notas[i];
    }
    return suma / n;
}

int contarAprobadas(float notas[], int n) {
    int contador = 0;
    for (int i = 0; i < n; i++) {
        if (notas[i] >= 3.0) {
            contador++;
        }
    }
    return contador;
}

int contarReprobadas(float notas[], int n) {
    int contador = 0;
    for (int i = 0; i < n; i++) {
        if (notas[i] < 3.0) {
            contador++;
        }
    }
    return contador;
}

void mostrarReporte(float mayor, float menor, float promedio, int aprobadas, int reprobadas) {
    cout << "\nReporte final\n";
    cout << "Nota mayor: " << mayor << endl;
    cout << "Nota menor: " << menor << endl;
    cout << "Promedio: " << promedio << endl;
    cout << "Aprobadas: " << aprobadas << endl;
    cout << "Reprobadas: " << reprobadas << endl;
}
